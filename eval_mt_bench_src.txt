1: """MT-Bench evaluation: answer generation and LLM-as-judge scoring.
2: 
3: Usage:
4:   # Generate answers
5:   python eval_mt_bench.py --model Qwen/Qwen2.5-7B-Instruct \
6:       --question_file dataset/eval/mt_bench/question.jsonl \
7:       --output_file results/mt_bench/Qwen2.5-7B-Instruct/answers.jsonl
8: 
9:   # Judge answers
10:   python eval_mt_bench.py --mode judge \
11:       --judge_model Qwen/Qwen2.5-7B-Instruct \
12:       --answer_file results/mt_bench/MODEL/answers.jsonl \
13:       --question_file dataset/eval/mt_bench/question.jsonl \
14:       --output_file results/mt_bench/MODEL/judgments.jsonl
15: """
16: 
17: import argparse
18: import json
19: import os
20: import torch
21: from tqdm import tqdm
22: from transformers import AutoTokenizer, AutoModelForCausalLM
23: from peft import PeftModel
24: import torch.nn as nn
25: 
26: 
27: class BinaryGate(nn.Module):
28:     """Lightweight MLP gate: hidden_state → {use_base, use_lora}."""
29:     def __init__(self, hidden_dim, dropout=0.1):
30:         super().__init__()
31:         self.gate = nn.Sequential(
32:             nn.Linear(hidden_dim, 256),
33:             nn.GELU(),
34:             nn.Dropout(dropout),
35:             nn.Linear(256, 2),
36:         )
37: 
38:     def forward(self, hidden_states):
39:         return self.gate(hidden_states)
40: 
41:     def predict(self, hidden_states):
42:         logits = self.forward(hidden_states)
43:         return logits.argmax(dim=-1)
44: 
45: 
46: def load_questions(path):
47:     """Load MT-Bench questions from JSONL."""
48:     questions = []
49:     with open(path) as f:
50:         for line in f:
51:             questions.append(json.loads(line.strip()))
52:     return questions
53: 
54: 
55: def generate_answer(model, tokenizer, messages, max_new_tokens=1024):
56:     """Generate a model response for a conversation."""
57:     text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
58:     inputs = tokenizer(text, return_tensors="pt", truncation=True, max_length=4096).to(model.device)
59: 
60:     with torch.no_grad():
61:         outputs = model.generate(
62:             **inputs,
63:             max_new_tokens=max_new_tokens,
64:             temperature=0.7,
65:             top_p=0.9,
66:             do_sample=True,
67:             pad_token_id=tokenizer.eos_token_id,
68:         )
69: 
70:     response = tokenizer.decode(outputs[0][inputs.input_ids.shape[1]:], skip_special_tokens=True)
71:     return response.strip()

74: def run_generation(args):
75:     """Generate model answers for all MT-Bench questions (2 turns each)."""
76:     questions = load_questions(args.question_file)
77:     print(f"Loaded {len(questions)} questions")
78: 
79:     tokenizer = AutoTokenizer.from_pretrained(args.model, trust_remote_code=True)
80:     tokenizer.pad_token = tokenizer.eos_token
81: 
82:     model = AutoModelForCausalLM.from_pretrained(
83:         args.model, device_map="auto", torch_dtype=torch.bfloat16, trust_remote_code=True
84:     )
85: 
86:     gate = None
87:     if args.adapter_path:
88:         print(f"Loading LoRA adapter from: {args.adapter_path}")
89:         model = PeftModel.from_pretrained(model, args.adapter_path)
90: 
91:         # Load gate if available
92:         gate_path = args.gate_path
93:         if not gate_path:
94:             # Auto-detect: look for gate.pt in adapter parent directory
95:             candidate = os.path.join(os.path.dirname(args.adapter_path), "gate.pt")
96:             if os.path.exists(candidate):
97:                 gate_path = candidate
98:         if gate_path and os.path.exists(gate_path):
99:             print(f"Loading gate from: {gate_path}")
100:             hidden_dim = model.config.hidden_size
101:             gate = BinaryGate(hidden_dim)
102:             gate.load_state_dict(torch.load(gate_path, map_location="cpu", weights_only=True))
103:             gate = gate.to(next(model.parameters()).device).to(next(model.parameters()).dtype)
104:             gate.eval()
105:             print(f"Gate loaded. Will route per-query: 0=base, 1=LoRA")
106: 
107:     model.eval()
108: 
109:     results = []
110:     for q in tqdm(questions, desc="Generating answers"):
111:         qid = q["question_id"]
112:         category = q["category"]
113:         turns = q["turns"]
114: 
115:         # Turn 1
116:         messages = []
117:         if args.force_empty_system:
118:             # Inject an explicit empty system message to suppress baked-in defaults
119:             messages.append({"role": "system", "content": ""})
120:         elif args.system_prompt:
121:             messages.append({"role": "system", "content": args.system_prompt})
122:         messages.append({"role": "user", "content": turns[0]})
123: 
124:         # Gate-based routing: decide whether to use LoRA for this query
125:         if gate is not None:
126:             gate_input = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
127:             gate_ids = tokenizer(gate_input, return_tensors="pt", truncation=True, max_length=4096).input_ids.to(model.device)
128:             model.disable_adapter_layers()
129:             with torch.no_grad():
130:                 out = model(input_ids=gate_ids, output_hidden_states=True)
131:                 hidden = out.hidden_states[1]  # layer 1
132:                 last_tok = hidden[0, -1, :].unsqueeze(0)
133:                 decision = gate.predict(last_tok).item()
134:             if decision == 1:
135:                 model.enable_adapter_layers()
136:                 # LoRA ON for this query
137:             # else: adapter stays disabled (base model)
138: 
139:         answer1 = generate_answer(model, tokenizer, messages, args.max_new_tokens)
140: 
141:         # Turn 2 (if exists)
142:         answer2 = ""
143:         if len(turns) > 1:
144:             messages.append({"role": "assistant", "content": answer1})
145:             messages.append({"role": "user", "content": turns[1]})
146:             answer2 = generate_answer(model, tokenizer, messages, args.max_new_tokens)
147: 
148:         result = {
149:             "question_id": qid,
150:             "category": category,
151:             "model": args.model.split("/")[-1],
152:             "turns": turns,
153:             "answers": [answer1, answer2] if answer2 else [answer1],
154:         }
155:         results.append(result)
156: 
157:     os.makedirs(os.path.dirname(args.output_file), exist_ok=True)
158:     with open(args.output_file, "w") as f:
159:         for r in results:
160:             f.write(json.dumps(r) + "\n")
161:     print(f"Saved {len(results)} answers to {args.output_file}")
162: 
163:     # Print per-category stats
164:     cats = {}
165:     for r in results:
166:         cats.setdefault(r["category"], []).append(r)
167:     print("\nPer-category question counts:")
168:     for cat in sorted(cats):
169:         print(f"  {cat}: {len(cats[cat])}")
170:

172: JUDGE_PROMPT = """Please act as an impartial judge and evaluate the quality of the response provided by an AI assistant to the user question displayed below. Your evaluation should consider factors including helpfulness, relevance, accuracy, depth, creativity, and level of detail of the response. Begin your evaluation by providing a short explanation. Be as objective as possible. After providing your explanation, output your rating on a scale of 1 to 10 by strictly following this format: "[[rating]]", for example: "Rating: [[5]]".
173: 
174: [Question]
175: {question}
176: 
177: [The Start of Assistant's Answer]
178: {answer}
179: [The End of Assistant's Answer]"""
180:

182: def run_judging(args):
183:     """Judge model answers using an LLM judge."""
184:     questions = {q["question_id"]: q for q in load_questions(args.question_file)}
185: 
186:     answers = []
187:     with open(args.answer_file) as f:
188:         for line in f:
189:             answers.append(json.loads(line.strip()))
190:     print(f"Loaded {len(answers)} answers to judge")
191: 
192:     tokenizer = AutoTokenizer.from_pretrained(args.judge_model, trust_remote_code=True)
193:     tokenizer.pad_token = tokenizer.eos_token
194: 
195:     judge = AutoModelForCausalLM.from_pretrained(
196:         args.judge_model, device_map="auto", torch_dtype=torch.bfloat16, trust_remote_code=True
197:     )
198:     judge.eval()
199: 
200:     judgments = []
201:     for ans in tqdm(answers, desc="Judging"):
202:         qid = ans["question_id"]
203:         q = questions[qid]
204:         category = ans["category"]
205: 
206:         # Judge each turn
207:         turn_scores = []
208:         for turn_idx, (question_text, answer_text) in enumerate(zip(q["turns"], ans["answers"])):
209:             prompt = JUDGE_PROMPT.format(question=question_text, answer=answer_text)
210:             messages = [{"role": "user", "content": prompt}]
211:             judgment_text = generate_answer(judge, tokenizer, messages, max_new_tokens=512)
212: 
213:             # Extract score
214:             score = extract_score(judgment_text)
215:             turn_scores.append(score)
216: 
217:         judgment = {
218:             "question_id": qid,
219:             "category": category,
220:             "model": ans["model"],
221:             "judge": args.judge_model.split("/")[-1],
222:             "turn_scores": turn_scores,
223:             "avg_score": sum(turn_scores) / len(turn_scores) if turn_scores else 0,
224:         }
225:         judgments.append(judgment)
226: 
227:     os.makedirs(os.path.dirname(args.output_file), exist_ok=True)
228:     with open(args.output_file, "w") as f:
229:         for j in judgments:
230:             f.write(json.dumps(j) + "\n")
231: 
232:     # Print per-category summary
233:     cat_scores = {}
234:     for j in judgments:
235:         cat_scores.setdefault(j["category"], []).append(j["avg_score"])
236: 
237:     print(f"\n{'Category':<20} {'Avg Score':>10} {'Count':>6}")
238:     print("-" * 40)
239:     all_scores = []
240:     for cat in sorted(cat_scores):
241:         scores = cat_scores[cat]
242:         avg = sum(scores) / len(scores)
243:         all_scores.extend(scores)
244:         print(f"{cat:<20} {avg:>10.2f} {len(scores):>6}")
245:     print("-" * 40)
246:     print(f"{'Overall':<20} {sum(all_scores)/len(all_scores):>10.2f} {len(all_scores):>6}")
247: 
248:     # Save summary
249:     summary = {
250:         "model": judgments[0]["model"] if judgments else "unknown",
251:         "judge": args.judge_model.split("/")[-1],
252:         "overall": sum(all_scores) / len(all_scores) if all_scores else 0,
253:         "per_category": {cat: sum(s)/len(s) for cat, s in cat_scores.items()},
254:         "n_questions": len(judgments),
255:     }
256:     summary_path = os.path.join(os.path.dirname(args.output_file), "mt_bench_summary.json")
257:     with open(summary_path, "w") as f:
258:         json.dump(summary, f, indent=2)
259:     print(f"\nSaved summary to {summary_path}")

262: def extract_score(text):
263:     """Extract numerical score from judge output."""
264:     import re
265:     # Look for [[N]] pattern
266:     match = re.search(r'\[\[(\d+(?:\.\d+)?)\]\]', text)
267:     if match:
268:         return float(match.group(1))
269:     # Fallback: look for "Rating: N"
270:     match = re.search(r'[Rr]ating:\s*(\d+(?:\.\d+)?)', text)
271:     if match:
272:         return float(match.group(1))
273:     # Last resort: look for any standalone number 1-10
274:     match = re.search(r'\b(\d+)\b', text)
275:     if match:
276:         score = int(match.group(1))
277:         if 1 <= score <= 10:
278:             return float(score)
279:     return 5.0  # default if extraction fails
